### `schemas.py` (build the Track Update schema the OSHConnect way)
```python id="jdax04"
from oshconnect.swe_components import (
    DataRecordSchema, TimeSchema, CountSchema, QuantitySchema, TextSchema
)
from oshconnect.api_utils import UCUMCode

def build_track_update_schema():
    # MUST be first for OSHConnect/OSH ([docs.opensensorhub.org](https://docs.opensensorhub.org/docs/osh-connect/getting-started/usage))
    time_schema = TimeSchema(
        name="timestamp",
        label="Epoch Timestamp",
        definition="https://os4csapi.org/def/odas/time/epochSeconds",
        referenceTime="1970-01-01T00:00:00Z",
        uom=UCUMCode(code="s", label="seconds"),
    )

    return DataRecordSchema(
        name="odas-track-update",
        label="ODAS Track Update (Normalized)",
        definition="https://os4csapi.org/def/odas/track/updateRecordOSH",
        fields=[
            time_schema,
            CountSchema(name="odasTimeStamp", label="ODAS timeStamp", definition="https://os4csapi.org/def/odas/frameIndex"),
            CountSchema(name="id", label="Track ID", definition="https://os4csapi.org/def/odas/trackId"),
            TextSchema(name="tag", label="Track tag", definition="https://os4csapi.org/def/odas/trackTag"),
            QuantitySchema(name="x", label="Direction cosine X", definition="https://os4csapi.org/def/odas/directionCosineX", uom=UCUMCode(code="1", label="1")),
            QuantitySchema(name="y", label="Direction cosine Y", definition="https://os4csapi.org/def/odas/directionCosineY", uom=UCUMCode(code="1", label="1")),
            QuantitySchema(name="z", label="Direction cosine Z", definition="https://os4csapi.org/def/odas/directionCosineZ", uom=UCUMCode(code="1", label="1")),
            QuantitySchema(name="activity", label="Activity (ODAS metric)", definition="https://os4csapi.org/def/odas/trackActivity", uom=UCUMCode(code="1", label="1")),
            QuantitySchema(name="azimuth", label="Azimuth (deg)", definition="https://os4csapi.org/def/odas/azimuth", uom=UCUMCode(code="deg", label="degrees")),
            QuantitySchema(name="elevation", label="Elevation (deg)", definition="https://os4csapi.org/def/odas/elevation", uom=UCUMCode(code="deg", label="degrees")),
        ],
    )
```

### `bootstrap.py` (connect + create systems + datastreams)
```python id="b7nlac"
from oshconnect.oshconnectapi import OSHConnect, TemporalModes
from oshconnect.streamableresource import Node, System
from oshconnect.csapi4py.constants import APIResourceTypes, ContentTypes
from oshconnect.timemanagement import TimeInstant, TimePeriod, TimeUtils

from .schemas import build_track_update_schema

def ensure_system_geojson(node: Node, *, uid: str, name: str, lon: float, lat: float) -> str:
    api = node.get_api_helper()

    # Try find by listing systems and matching properties.uid (idempotent)
    res = api.retrieve_resource(APIResourceTypes.SYSTEM, req_headers={"Accept": ContentTypes.GEO_JSON.value})
    if res.ok:
        for item in res.json().get("items", []):
            props = item.get("properties", {})
            if props.get("uid") == uid:
                return item.get("id") or item.get("properties", {}).get("id") or item.get("properties", {}).get("system_id")

    payload = {
        "type": "Feature",
        "geometry": {"type": "Point", "coordinates": [lon, lat]},
        "properties": {
            "featureType": "sosa:Sensor",
            "uid": uid,
            "name": name,
            "description": "ODAS mic array (simulated) — Ft Huachuca NE range area",
        }
    }

    create = api.create_resource(
        APIResourceTypes.SYSTEM,
        payload,
        req_headers={"Content-Type": ContentTypes.GEO_JSON.value, "Accept": ContentTypes.GEO_JSON.value},
    )
    if not create.ok:
        raise RuntimeError(f"Failed to create system {name}: {create.status_code} {create.text}")
    return create.headers["Location"].split("/")[-1]

def bootstrap(node_address="45.55.99.236", node_port=8080, username="ogc", password="ogc"):
    app = OSHConnect(name="ODAS-MA-Sim", playback_mode=TemporalModes.REAL_TIME)
    node = Node(protocol="http", address=node_address, port=node_port, username=username, password=password, server_root="sensorhub")
    app.add_node(node)

    # Ft Huachuca “best judgment” emplacement (edit anytime)
    ma = {
        "MA-1": {"uid": "urn:os4csapi:system:odas:ma-1", "name": "ODAS Mic Array MA-1", "lat": 31.663006, "lon": -110.272897},
        "MA-2": {"uid": "urn:os4csapi:system:odas:ma-2", "name": "ODAS Mic Array MA-2", "lat": 31.655006, "lon": -110.257897},
        "MA-3": {"uid": "urn:os4csapi:system:odas:ma-3", "name": "ODAS Mic Array MA-3", "lat": 31.650006, "lon": -110.269897},
    }

    schema = build_track_update_schema()

    systems = {}
    for key, cfg in ma.items():
        sys_id = ensure_system_geojson(node, uid=cfg["uid"], name=cfg["name"], lon=cfg["lon"], lat=cfg["lat"])
        sys_obj = System(name=key.lower(), label=cfg["name"], urn=cfg["uid"], parent_node=node, resource_id=sys_id)
        systems[key] = sys_obj

        # Create datastream (TimeSchema first is already satisfied) ([docs.opensensorhub.org](https://docs.opensensorhub.org/docs/osh-connect/getting-started/usage))
        sys_obj.add_insert_datastream(schema)

    return app, node, systems
```

### `simulator.py` (publish ODAS-like track updates)
```python id="xo1b5i"
import math, random, time
from datetime import datetime, timezone
from oshconnect.timemanagement import TimeInstant

def dircos_from_bearing_elev(bearing_deg: float, elev_deg: float):
    # ENU convention: x=East, y=North is common; BUT we’re publishing azimuth/elevation as explicit fields
    # For direction cosines we’ll treat x/y as a unit circle in the horizontal plane.
    az = math.radians(bearing_deg)
    el = math.radians(elev_deg)
    x = math.cos(el) * math.cos(az)
    y = math.cos(el) * math.sin(az)
    z = math.sin(el)
    return x, y, z

def run_track_sim(datastream, hz=1.0):
    frame = 0
    track_id = 1
    bearing = 45.0
    elev = 10.0

    while True:
        now = TimeInstant.now_as_time_instant()
        frame += 1
        bearing = (bearing + 5.0) % 360.0
        elev = max(-20.0, min(20.0, elev + random.uniform(-1.0, 1.0)))
        x, y, z = dircos_from_bearing_elev(bearing, elev)

        obs = {
            "resultTime": now.get_iso_time(),        # required ([docs.opensensorhub.org](https://docs.opensensorhub.org/docs/osh-connect/getting-started/usage))
            "phenomenonTime": now.get_iso_time(),    # required ([docs.opensensorhub.org](https://docs.opensensorhub.org/docs/osh-connect/getting-started/usage))
            "result": {
                "timestamp": now.epoch_time,         # MUST be present and first in schema ([docs.opensensorhub.org](https://docs.opensensorhub.org/docs/osh-connect/getting-started/usage))
                "odasTimeStamp": frame,
                "id": track_id,
                "tag": "dynamic",
                "x": round(x, 3),
                "y": round(y, 3),
                "z": round(z, 3),
                "activity": round(random.uniform(0.0, 1.0), 3),
                "azimuth": round(bearing, 3),
                "elevation": round(elev, 3),
            }
        }
        datastream.insert_observation_dict(obs)
        time.sleep(1.0 / hz)
```
